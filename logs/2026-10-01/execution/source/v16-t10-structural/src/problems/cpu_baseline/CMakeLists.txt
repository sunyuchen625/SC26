find_package(HYPRE 2.33 CONFIG REQUIRED)
add_library(sc26_problem_dependencies INTERFACE)
target_link_libraries(sc26_problem_dependencies INTERFACE HYPRE::HYPRE)
math(EXPR SC26_PROBLEM_NUMBER "${SC26_PROBLEM}")
target_compile_definitions(sc26_problem_dependencies INTERFACE SC26_PROBLEM_NUMBER=${SC26_PROBLEM_NUMBER})
if(SC26_PROBLEM STREQUAL "05")
    set(SC26_FIELD_SPLIT_DEFAULT ON)
else()
    set(SC26_FIELD_SPLIT_DEFAULT OFF)
endif()
option(SC26_FIELD_SPLIT "Use pressure MGR for T5 or T7" ${SC26_FIELD_SPLIT_DEFAULT})
if(SC26_FIELD_SPLIT)
    if(NOT SC26_PROBLEM MATCHES "^(05|07)$")
        message(FATAL_ERROR "SC26_FIELD_SPLIT requires problem 05 or 07")
    endif()
    target_compile_definitions(sc26_problem_dependencies INTERFACE SC26_FIELD_SPLIT)
endif()
find_path(MKL_INCLUDE_DIR mkl.h HINTS ENV MKLROOT PATH_SUFFIXES include REQUIRED)
find_library(MKL_RT mkl_rt HINTS ENV MKLROOT PATH_SUFFIXES lib/intel64 REQUIRED)
target_include_directories(sc26_problem_dependencies INTERFACE "${MKL_INCLUDE_DIR}")
target_link_libraries(sc26_problem_dependencies INTERFACE "${MKL_RT}" pthread dl m)
